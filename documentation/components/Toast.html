import { Component } from '@angular/core';
import { CommonModule } from '@angular/common';
import { NotificacionService } from '../../services/notificacion';

@Component({
  selector: 'app-toast',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './toast.html',
  styleUrl: './toast.css'
})
export class Toast {
  constructor(public notificacionService: NotificacionService) {}

  trackById(index: number, notificacion: { id: number }): number {
    return notificacion.id;
  }
}
<div class="toast-container position-fixed top-0 end-0 p-3" style="z-index: 1080;">
  <div
    *ngFor="let notificacion of notificacionService.notificaciones$ | async; trackBy: trackById"
    class="toast show rounded-3 shadow-sm mb-2"
    role="alert"
    aria-live="assertive"
    aria-atomic="true">
    <div class="toast-body d-flex align-items-center justify-content-between"
      [class.text-bg-danger]="notificacion.tipo === 'error'"
      [class.text-bg-success]="notificacion.tipo === 'success'"
      [class.text-bg-info]="notificacion.tipo === 'info'">
      <span>{{ notificacion.mensaje }}</span>
      <button type="button" class="btn-close btn-close-white ms-3"
        (click)="notificacionService.cerrar(notificacion.id)"
        aria-label="Cerrar"></button>
    </div>
  </div>
</div>
